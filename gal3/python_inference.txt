    method string Python_String
    {

    }
    method string Python_Enquote [string Text]
    {
        Text = (i Python_Escape_Content Text);
        return (i Python_Choose_Quotes Text);
    }
    method string Python_Escape_Content [string Text]
    {
        if (Text contains '\')
        {
            replace Text '\' '\\';
        }
        if (Text contains [tab])
        {
            replace Text [tab] "\t";
        }
        if (Text contains [line])
        {
            replace Text [line] "\n";
        }
        string CR (Runtime:Chr 13);
        if (Text contains CR)
        {
            replace Text CR "\r";
        }
        return Text;
    }
    method string Python_Choose_Quotes [string Text]
    {
        return (append "'" Text "'" ) if (not (Text contains "'"));
        return (append '"' Text '"' ) if (not (Text contains '"'));
        return (append "'''" Text "'''" ) if (not (Text contains "'''"));
        return (append '"""' Text '"""' ) if (not (Text contains '"""'));
        replace Text "'" "\'";
        return (append "'''" Text "'''");
    }
    method string Python_Indent [string Text]
    {
        return '' if (isnull Text);
        dictionary Qend;
        Qend '"' d= '"';
        Qend "'" d= "'";
        Qend '`' d= '`';
        Qend '"""' d= '"""';
        Qend "'''" d= "'''";
        integer P;
        string C;
        string C3;
        string I '';
        flag Quote [false];
        string Q '';
        string Between [indent];
        integer E = ((length Text) - 1);
        integer Skip = 0;
        for P range 0 E
        {
            C = (substring Text P 1);
            C3 = (substring Text P 3);
            append I Between C;
            Between = '';
            if (Skip > 0)
            {
                decrement Skip;
            }
            else if Quote
            {
                if (C3 s= Q)
                {
                    Quote = [false];
                    Q = '';
                    Skip = ((length C3) - 1);
                }
                else if (C s= Q)
                {
                    Quote = [false];
                    Q = '';
                }
            }
            else if (C = [line])
            {
                Between = [indent];
            }
            else if (Qend dexists C3)
            {
                Quote = [true];
                Q = (dget Qend C3);
            }
            else if (Qend dexists C)
            {
                Quote = [true];
                Q = (dget Qend C);
            }
        }
        return I;
    }
    method void Python_Append_Flatten [object Arg, list Into]
    {
        comment `a nested append operation inlines its segments: embedding
        its finished f-string inside another f-string's braces would nest
        same-quote f-strings, which python 3.11 rejects`;
        forgive
        {
            object AC [Arg.Classification];
            if (AC is an Operation_Append)

                object M;
                for each M in [AC.Arguments]
                {
                    i Python_Append_Flatten M Into;
                }
                return;
            }
        }
        push Into Arg;
    }

    method string Python_Append [list Raw_Arguments]
    {

        object A0;
        for each A0 in Raw_Arguments

        }
        return '' if ((llength Arguments) = 0);
        if ((llength Arguments) = 1)
        {

            return (Arg.Python_String);
        }
        objects Arg Argc;
        string U;
        string Code '';
        flag Embed = [false];
        for each Arg in Arguments
        {
            Argc = [Arg.Classification];
            if (Argc is a Literal_Syntax)
            {
                if (not (defined [Argc.Unquoted]))
                {
                    Embed = [true];
                }
            }
            else if (Argc is a Token_Quote)
            {
            }
            else if (Argc is a Token_Number)
            {
            }
            else
            {
                Embed = [true];
            }
        }
        comment `literal segments are escaped individually; embedded expression
        segments are code and must pass through verbatim, so the final quote
        wrap must not re-escape the assembled body`;
        for each Arg in Arguments
        {
            Argc = [Arg.Classification];
            if ((Argc is a Literal_Syntax) and (defined [Argc.Unquoted]))
            {
                U = (i Python_Escape_Content [Argc.Unquoted]);
                if Embed
                {
                    if (U contains '{')
                    {
                        replace U '{' '{{';
                    }
                    if (U contains '}')
                    {
                        replace U '}' '}}';
                    }
                }
                append Code U;
            }
            else if (Argc is a Token_Quote)
            {
                U = (i Python_Escape_Content [Argc.Unquoted]);
                if Embed
                {
                    if (U contains '{')
                    {
                        replace U '{' '{{';
                    }
                    if (U contains '}')
                    {
                        replace U '}' '}}';
                    }
                }
                append Code U;
            }
            else if (Argc is a Token_Number)
            {
                append Code [Argc.Python];
            }
            else
            {
                append Code '{ ' [Argc.Python] ' }';
            }
        }
        Code = (i Python_Choose_Quotes Code);
        if Embed
        {
            Code = (append "f" Code);
        }
        return Code;
    }

class set Operation
{
    method string Python_Numeric_Operand [object Op]
    {
        comment `a numeric operation converts a string-typed operand:
        int when the surrounding usage is integer, float otherwise. python
        would raise a TypeError on the unconverted form. only a definite
        string type converts - variant and unknown stay untouched.`;
        string Code [Op.Python];
        if ((i Python_Arg_Precedence Op) > [my Python_Precedence])
        {
            Code = (append '(' Code ')');
        }
        if ((i Python_Arg_Data_Type Op) s= 'string')
        {
            string Conv 'float';
            if ([my Usage] s= 'integer')
            {
                Conv = 'int';
            }
            Code = (append Conv '(' Code ')');
        }
        return Code;
    }

    method string Python_Numeric_Infix [object Op1, object Op2, string Infix]
    {
        string Code (i Python_Numeric_Operand Op1);
        append Code Infix (i Python_Numeric_Operand Op2);
        return Code;
    }

    method string Python_Numeric_Repeating [string Oper]
    {
        comment `the numeric chain converts string-typed operands the same
        way Python_Numeric_Operand does, in every position of the chain`;
        object Argument;
        string Arg;
        string Code '';
        flag First [true];
        for each Argument in [my Arguments]
        {
            Arg = [Argument.Python];
            integer P (i Python_Arg_Precedence Argument);
            if (P > [my Python_Precedence])
            {
                Arg = (append '(' Arg ')');
            }
            else if ((P = [my Python_Precedence]) and (not First))
            {
                Arg = (append '(' Arg ')');
            }
            if ((i Python_Arg_Data_Type Argument) s= 'string')
            {
                string Conv 'float';
                if ([my Usage] s= 'integer')
                {
                    Conv = 'int';
                }
                Arg = (append Conv '(' Arg ')');
            }
            if (Code sgt '')
            {
                append Code Oper;
            }
            append Code Arg;
            First = [false];
        }
        return Code;
    }

    method string Python_Infix [object Op1, object Op2, string Infix]
    {
        string Left [Op1.Python];
        string Right [Op2.Python];
        if ((i Python_Arg_Precedence Op1) > [my Python_Precedence])
        {
            Left = (append '(' Left ')');
        }
        if ((i Python_Arg_Precedence Op2) >= [my Python_Precedence])
        {
            comment `the right operand needs parentheses at equal precedence
            too, because infix operators associate left: 10 - (4 - 3)`;
            Right = (append '(' Right ')');
        }
        string Code Left Infix Right;
        return Code;
    }
    method string Python_Unary [object Op1, string Oper]
    {
        string Arg [Op1.Python];
        if ((i Python_Arg_Precedence Op1) >= [my Python_Precedence])
        {
            Arg = (append '(' Arg ')');
        }
        string Code Oper Arg;
        return Code;
    }
}

class Python_Compiler is Fallback_Compiler
{
    constructor
    {
        my Generator = (new Python_Generator [self]);
    }

    method void Print_Errors
    {
        object E;
        for each E in [my Elements]
        {
            if (not ([E.Error] is null))
            {
                say [line] 
                    [class name [E.Classification]] 
                    ': ' 
                    (substring (E.Get_Input) 0 80);
                say "ERROR: " [E.Error];
            }
        }
    }
    method number Infer_Code
    {
        verbosity;
        verbose "Python_Compiler.Infer_Code: starting code generation";
        number Status 1;

        [my Generator Elements] = [my Elements];
        Status = ([my Generator].Infer_Gal);
        if (not Status)
        {
            i Print_Errors;
            say "Error in Python Inference, inferring gal";
            return 0;
        }
        Status = ([my Generator].Infer_Fallback);
        if (not Status)
        {
            i Print_Errors;
            say "Error in Python Inference, inferring fallback";
            return 0;
        }

        verbose "Python_Compiler.Infer_Code: calling Generator.Infer_Python";

        # 'Call Infer_Python to generate the code';
        [my Generator].Infer_Python;

        # 'Get the top-level program element';
        object Top = (llast [my Elements]);

        verbose "Python_Compiler.Infer_Code: ensuring Top element has python code";

        verbose (Top.To_String);
        try
        {
            if ([Top.Python] is null)
            {
                Top.Infer_Python;
            }
        }
        catch Error
        {
            i Print_Errors;
            say "Error in Python Inference, inferring python";
            return 0;
        }

        string Code = [Top.Python];
        my Python = Code;

        verbose "Python_Compiler.Infer_Code: successfully generated python code";

        return 1;
    }
}
