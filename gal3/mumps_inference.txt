flowerbox 'mumps inference: a mumps generator growing from the hello contract';

comment `
The mumps backend, started at hello coverage and growing by the
five-languages contract, then the java-port playbook (docs/
mumps_design.md). It must compile what it claims and fail loudly beyond
that, per the c_inference.txt philosophy.

mumps is the persistent-database point on the language spectrum: a
module becomes one routine, and (as coverage grows) each class becomes
one routine of flattened labels via the inheritance inference, objects
live in globals, and collections are sparse subscript trees. gal2's
proven generator (gal2/src/Mumps.gal) is the reference these bodies
port from.

emission conventions: labels start in column one, command lines indent
(spaces are fine for yottadb); strings quote with doubled ""; say maps
to use $principal write ...,!; a module routine header does main and
quits before the labels, so falling in from the top never executes a
formallist label.
`;

class set Element
{
    property string Mumps;
    property string Mumps_Statements '';
    property string Mumps_End_Label '';
    property string Mumps_Etrap_Saved '';
    property dictionary Mumps_Locals;

    method object Mumps_Owner_Method
    {
        comment `the enclosing frame: a method, constructor or main, which
        is where locals are newed`;
        if ([my Classification] is a Function_Statement)
        {
            return [my Classification];
        }
        if ([my Classification] is a Statement_Constructor)
        {
            return [my Classification];
        }
        if ([my Classification] is a Statement_Main)
        {
            return [my Classification];
        }
        forgive
        {
            return ([my Parent].Mumps_Owner_Method);
        }
        return [null];
    }

    method string Mumps_Declare [string Name]
    {
        comment `mumps NEW pushes a save entry that only pops when the
        frame quits, so a NEW inside a goto-loop leaks one entry per
        iteration and exhausts the stack on a big enough program. every
        local therefore registers here and is newed once in its method's
        prologue - which also matches python's function-scoped locals.
        generation is post-order, so a method's body registers before the
        method assembles its prologue. a declaration with no enclosing
        frame (module top level) keeps an inline new`;
        forgive
        {
            object M (i Mumps_Owner_Method);
            if (not (M is null))
            {
                [M.Mumps_Locals] Name d= 'y';
                return '';
            }
        }
        return (append 'new ' Name ' ');
    }

    method string Mumps_Cond [object Op]
    {
        comment `python truthiness, which mumps does not share: mumps
        reads a string numerically, so an object handle, a class name or
        any non-numeric text evaluates to zero - false - where python
        calls it true. a definitely-numeric operand can be tested
        directly; everything else goes through the runtime`;
        string Code [Op.Mumps];
        string T (i Mumps_Arg_Data_Type Op);
        if (T s= 'flag')
        {
            return Code;
        }
        if (T s= 'integer')
        {
            return Code;
        }
        if (T s= 'number')
        {
            return Code;
        }
        return (append '$$truthy^galrt(' Code ')');
    }

    method string Mumps_Trap_Value
    {
        comment `the $etrap in force around this statement: the nearest
        enclosing guard's trap literal, or the frame's inherited %etin.
        a guard restores this at its landing label instead of saving the
        old value in a local, so no per-guard new is needed`;
        forgive
        {
            return ([my Parent].Mumps_Trap_Here);
        }
        return '%etin';
    }

    method string Mumps_Trap_Here
    {
        comment `walking outward: a guard answers with its own trap, a
        frame boundary ends the walk. a catch body's parent is the catch
        statement, whose parent is the try's enclosing scope, so the try
        it belongs to is correctly skipped`;
        if ([my Classification] is a Statement_Forgive)
        {
            return (append '"quit:$estack  goto L' [my Start_Position] 'H"');
        }
        if ([my Classification] is a Statement_Try)
        {
            return (append '"quit:$estack  goto L' [my Start_Position] 'H"');
        }
        if ([my Classification] is a Statement_Defined)
        {
            return (append '"quit:$estack  goto L' [my Start_Position] 'H"');
        }
        if ([my Classification] is a Statement_Undefined)
        {
            return (append '"quit:$estack  goto L' [my Start_Position] 'H"');
        }
        if ([my Classification] is a Function_Statement)
        {
            return '%etin';
        }
        if ([my Classification] is a Statement_Constructor)
        {
            return '%etin';
        }
        if ([my Classification] is a Statement_Main)
        {
            return '%etin';
        }
        forgive
        {
            return ([my Parent].Mumps_Trap_Here);
        }
        return '%etin';
    }

    method string Mumps_Hoist
    {
        comment `the frame prologue: one new for every registered local,
        plus the error-handling context. $estack is newed here so the
        guard idiom quit:$estack sees zero at this frame's own level.
        a name that is also a formal parameter must NOT be newed - that
        would wipe the value the caller passed`;
        string Names '%etin';
        dictionary Params;
        forgive
        {
            object PA;
            for each PA in [my Listargs]
            {
                string PN [PA.Mumps];
                Params PN d= 'y';
            }
        }
        forgive
        {
            iterate [my Mumps_Locals] DN DV
            {
                if (not (Params dexists DN))
                {
                    append Names ',' DN;
                }
            }
        }
        return (append [indent] 'new ' Names ' set %etin=$etrap' [line]
            [indent] 'new $etrap,$estack set $etrap=%etin' [line]);
    }

    method string Mumps_Quote [string Unquoted]
    {
        comment `mumps strings double their quotes; a real newline can
        only be expressed by concatenating $char(10)`;
        string Text Unquoted;
        if (Text contains '"')
        {
            replace Text '"' '""';
        }
        if (Text contains [line])
        {
            replace Text [line] '"_$char(10)_"';
        }
        return (append '"' Text '"');
    }

    method string Mumps_Block
    {
        comment `the block's code, or empty for a bodyless statement;
        probed through object variables because a string declaration
        would coerce None to the text 'None' under the python host`;
        forgive
        {
            object BE [my Block];
            if (BE is an Element)
            {
                object BV [BE.Mumps];
                if (not (BV is null))
                {
                    return BV;
                }
            }
        }
        return '';
    }

    method string Mumps_Arg_Data_Type [object Arg]
    {
        forgive
        {
            object AC [Arg.Classification];
            object DT [AC.Data_Type];
            if (not (DT is an Element))
            {
                if (not (DT is null))
                {
                    return DT;
                }
            }
        }
        return '';
    }

    method string Mumps_Deref [string Recv, string Sub]
    {
        comment `collections are matrix-node handles (reference
        semantics, as python): the receiver expression evaluates to a
        handle, and access double-indirects through it`;
        if (Recv is null)
        {
            error 'mumps: dereferencing an ungenerated receiver';
        }
        return (append '@(' Recv ')@(' Sub ')');
    }

    method string Mumps_Alloc
    {
        comment `a fresh matrix node's handle`;
        return '$name(@%matrix@($increment(@%matrix)))';
    }

    method string Mumps_Subscripted [string Ref, string Sub]
    {
        comment `extend a collection reference with one more subscript:
        a property node @%self@("P") gains ,Sub inside its parentheses,
        a plain local D becomes D(Sub)`;
        if (Ref is null)
        {
            error 'mumps: subscripting an ungenerated reference';
        }
        integer L (length Ref);
        if (L > 0)
        {
            string Last (substring Ref (L - 1) 1);
            if (Last s= ')')
            {
                return (append (substring Ref 0 (L - 1)) ',' Sub ')');
            }
        }
        return (append Ref '(' Sub ')');
    }

    method string Mumps_Break_Label
    {
        if ([my Classification] is an Iteration_Statement)
        {
            return (append 'L' [my Start_Position] 'K');
        }
        forgive
        {
            return ([my Parent].Mumps_Break_Label);
        }
        return 'LBREAKLOST';
    }

    method string Mumps_Continue_Label
    {
        if ([my Classification] is an Iteration_Statement)
        {
            return (append 'L' [my Start_Position] 'C');
        }
        forgive
        {
            return ([my Parent].Mumps_Continue_Label);
        }
        return 'LCONTINUELOST';
    }

    method string Mumps_Args [string Separator]
    {
        object Arg;
        string Args_Mumps '';
        string Between '';
        for each Arg in [my Listargs]
        {
            append Args_Mumps Between [Arg.Mumps];
            = Between Separator;
        }
        return Args_Mumps;
    }

    method string Mumps_Arguments [string Separator]
    {
        object Arg;
        string Args_Mumps '';
        string Between '';
        for each Arg in [my Arguments]
        {
            append Args_Mumps Between [Arg.Mumps];
            = Between Separator;
        }
        return Args_Mumps;
    }
}

generator Mumps Fallback
{
    english 'generate mumps code for each language element, using the inferred data types.';

    generate Element
    {
        i = '';
    }

    generate Statement
    {
        error 'no mumps generator for statement ' [class name];
    }

    generate Operation
    {
        error 'no mumps generator for operation ' [class name];
    }

    generate Syntax
    {
        error 'no mumps generator for syntax ' [class name];
    }

    generate Token
    {
        comment `mumps identifiers allow no underscores; names strip them
        (the gal2 convention). literals go through Token_Quote and keep
        theirs`;
        string T [my Text];
        if (T contains '_')
        {
            replace T '_' '';
        }
        i = T;
    }

    generate Token_Quote
    {
        i = (i Mumps_Quote [my Unquoted]);
    }

    generate Syntax_True
    {
        i = '1';
    }

    generate Syntax_False
    {
        i = '0';
    }

    generate Syntax_Line
    {
        my Unquoted = [line];
        i = '$char(10)';
    }

    generate Program
    {
        string Code '';
        object Stmt;
        for each Stmt in [my Statements]
        {
            undefined [Stmt.Mumps]
            {
                Stmt.Infer_Mumps;
            }
            string S [Stmt.Mumps];
            if (S sgt '')
            {
                append Code S;
            }
        }
        i = Code;
    }

    generate Block
    {
        comment `a mumps block is just its command lines; the statements
        already carry their own indentation, and the enclosing label
        supplies entry and quit`;
        string Statements '';
        object Line;
        for each Line in [my Statements]
        {
            string S [Line.Mumps];
            if (S sgt '')
            {
                append Statements S;
            }
        }
        my Mumps_Statements = Statements;
        i = Statements;
    }

    generate Statement_Comment
    {
        i = '';
    }

    generate Statement_Flowerbox
    {
        i = '';
    }

    generate Statement_Module
    {
        comment `one routine per module: a header label, an immediate do
        main and quit so top entry never falls into a formallist label,
        then the labels the module body produced`;
        string Name [my Name Mumps];
        string Body '';
        string Labels '';
        string Class_Sections '';
        object Stmt;
        forgive
        {
            for each Stmt in [my Block Statements]
            {
                comment `class sections open their own routines, so they
                must follow every module-level label (main included) or
                those labels land in the wrong file`;
                object SC2 [Stmt.Classification];
                object SV [Stmt.Mumps];
                if (SV is null)
                {
                }
                else if (SC2 is a Class_Statement)
                {
                    append Class_Sections SV;
                }
                else
                {
                    if (SV sgt '')
                    {
                        append Labels SV;
                    }
                }
            }
        }
        append Labels Class_Sections;
        string Inits '';
        flag Clash [false];
        forgive
        {
            object CS;
            for each CS in [my Block Statements]
            {
                forgive
                {
                    object CC2 [CS.Classification];
                    if (CC2 is a Class_Statement)
                    {
                        string CN2 [CC2.Class_Name Mumps];
                        replace CN2 '_' '';
                        if (CN2 s= Name)
                        {
                            Clash = [true];
                        }
                        append Inits [indent] 'do %init^' CN2 [line];
                    }
                }
            }
        }
        if Clash
        {
            comment `a class may not share the module routine's name; the
            module steps aside, as the java backend's public class does`;
            Name = (append Name 'Program');
        }
        string Code ';;;ROUTINE ' Name [line]
            Name ' ; gal ; ' Name ' module.' [line]
            [indent] 'do init^galrt' [line]
            Inits
            [indent] 'do main()' [line]
            [indent] 'quit' [line]
            Labels;
        i = Code;
    }

    generate Statement_Main
    {
        comment `mumps has no argv: a main with parameters takes them as
        space-separated pieces of $zcmdline after the routine name`;
        string Params '';
        integer N 1;
        object Arg;
        forgive
        {
            for each Arg in [my Listargs]
            {
                append Params [indent] 'new ' [Arg.Mumps] ' set ' [Arg.Mumps] '=$piece($zcmdline," ",' N ')' [line];
                N = (N + 1);
            }
        }
        string Code 'main() ; main program' [line]
            (i Mumps_Hoist)
            Params
            (i Mumps_Block)
            [indent] 'quit' [line];
        i = Code;
    }

    generate Statement_Say
    {
        i = (append [indent] 'use $principal write ' (i Mumps_Arguments ',') ',!' [line]);
    }

    generate Statement_Debug
    {
        comment `the %debug routine (tools/yottadb/routines/_debug.m,
        installed in ydb_routines) is the mumps equivalent of zdebug:
        an interactive shell with step and watch. gal2 emitted 'break
        %debug'; entering the shell directly is the same intent`;
        i = (append [indent] 'do ^%debug' [line]);
    }

    generate Statement_Error
    {
        comment `the gal2 form: a user $ecode carrying the message, which
        fires the nearest $etrap`;
        i = (append [indent] 'set $ecode="U:"_' (i Mumps_Arguments '_') [line]);
    }

    generate Statement_Forgive
    {
        comment `the gal2 $etrap pattern: trap to a local label, clear the
        error state, fall through. the outer $etrap is saved and restored
        at the landing label: new alone only restores at frame pop, so
        without the restore an error AFTER the guard would goto backward
        into the label and loop forever`;
        string Catch_Label 'L' [my Start_Position] 'H';
        string Code
            [indent] 'set $etrap="quit:$estack  goto ' Catch_Label '"' [line]
            (i Mumps_Block)
            Catch_Label ' ; forgiven' [line]
            [indent] 'set $etrap=' (i Mumps_Trap_Value) ',%ecode=$zstatus,$ecode="",$zstatus=""' [line];
        i = Code;
    }

    generate Statement_Try
    {
        comment `try traps to the catch label its own emission ends with;
        the catch statement's text follows and closes with the shared end
        label, handed over through the Next_Catch link`;
        string End 'L' [my Start_Position] 'E';
        string Catch_Label 'L' [my Start_Position] 'H';
        string Restore (i Mumps_Trap_Value);
        forgive
        {
            object N [my Next_Catch];
            if (N is a Catch_Statement)
            {
                [N.Mumps_End_Label] = End;
                [N.Mumps_Etrap_Saved] = Restore;
            }
        }
        string Code
            [indent] 'set $etrap="quit:$estack  goto ' Catch_Label '"' [line]
            (i Mumps_Block)
            [indent] 'goto ' End [line]
            Catch_Label ' ; try catch' [line];
        i = Code;
    }

    generate Statement_Catch
    {
        string Variable_Code '';
        forgive
        {
            if (not ([my Error] is null))
            {
                Variable_Code = (append [indent] 'set ' [my Error Mumps] '=%ecode' [line]);
            }
        }
        comment `restore the try's saved outer trap first, so an error in
        the catch body escapes outward instead of looping back to the
        catch label; the restore repeats after the end label for the
        success path, which jumps straight there`;
        string Code
            [indent] 'set $etrap=' [my Mumps_Etrap_Saved] ',%ecode=$zstatus,$ecode="",$zstatus=""' [line]
            Variable_Code
            (i Mumps_Block)
            [my Mumps_End_Label] ' ; end try' [line]
            [indent] 'set $etrap=' [my Mumps_Etrap_Saved] [line];
        i = Code;
    }

    generate Statement_Return
    {
        comment `the gal2 dual-context quit: an extrinsic call takes the
        value, a do-call quits plain`;
        string Value '""';
        forgive
        {
            if (defined [my Value])
            {
                Value = [my Value Mumps];
            }
        }
        string Code
            [indent] 'set %("return")=' Value [line]
            [indent] 'quit:$quit %("return") quit' [line];
        i = Code;
    }

    generate Statement_Return_If
    {
        string Value '""';
        forgive
        {
            if (defined [my Value])
            {
                Value = [my Value Mumps];
            }
        }
        string Skip 'L' [my Start_Position] 'S';
        string Code
            [indent] 'goto:' "'" '(' (i Mumps_Cond [my Condition]) ') ' Skip [line]
            [indent] 'set %("return")=' Value [line]
            [indent] 'quit:$quit %("return") quit' [line]
            Skip ' ; return-if skipped' [line];
        i = Code;
    }

    generate Statement_If_Return
    {
        string Value '""';
        forgive
        {
            if (defined [my Value])
            {
                Value = [my Value Mumps];
            }
        }
        string Skip 'L' [my Start_Position] 'S';
        string Code
            [indent] 'goto:' "'" '(' (i Mumps_Cond [my Condition]) ') ' Skip [line]
            [indent] 'set %("return")=' Value [line]
            [indent] 'quit:$quit %("return") quit' [line]
            Skip ' ; return-if skipped' [line];
        i = Code;
    }

    generate Statement_I
    {
        comment `flattening makes a direct in-routine call virtually
        correct, and the Compiled_Methods table says statically whether
        the label exists; a method the table lacks (python's tolerated
        missing attributes) dispatches through the runtime instead, so
        zlink never sees a dead reference`;
        string Args (i Mumps_Args ',');
        string M [my Method Mumps];
        string MN M;
        replace MN '_' '';
        flag Known [false];
        forgive
        {
            object OC (i Get_Owner_Class);
            if ([OC.Compiled_Methods] dexists M)
            {
                Known = [true];
            }
        }
        string Code '';
        if Known
        {
            Code = (append [indent] 'do ' MN '(%self');
            if (Args sgt '')
            {
                append Code ',' Args;
            }
            append Code ')' [line];
        }
        else
        {
            Code = (append [indent] 'do dispatch' (llength [my Listargs]) '^galrt(%self,' (i Mumps_Quote MN));
            if (Args sgt '')
            {
                append Code ',' Args;
            }
            append Code ')' [line];
        }
        i = Code;
    }

    generate Operation_I
    {
        string Args (i Mumps_Args ',');
        string M [my Method Mumps];
        string MN M;
        replace MN '_' '';
        flag Known [false];
        forgive
        {
            object OC (i Get_Owner_Class);
            if ([OC.Compiled_Methods] dexists M)
            {
                Known = [true];
            }
        }
        string Code '';
        if Known
        {
            Code = (append '$$' MN '(%self');
            if (Args sgt '')
            {
                append Code ',' Args;
            }
            append Code ')';
        }
        else
        {
            Code = (append '$$dispatch' (llength [my Listargs]) '^galrt(%self,' (i Mumps_Quote MN));
            if (Args sgt '')
            {
                append Code ',' Args;
            }
            append Code ')';
        }
        i = Code;
    }

    generate Syntax_Property
    {
        comment `a property chain on a variable holding an object handle,
        exactly the Syntax_My shape with the variable as the base`;
        string Code '';
        flag First [true];
        object Arg;
        for each Arg in [my Arguments]
        {
            if First
            {
                Code = [Arg.Mumps];
                First = [false];
            }
            else
            {
                Code = (append '@' Code '@(' (i Mumps_Quote [Arg.Mumps]) ')');
            }
        }
        i = Code;
    }

    generate Statement_Propset
    {
        i = (append [indent] 'set @' [my Object Mumps] '@(' (i Mumps_Quote [my Property Mumps]) ')=' [my Value Mumps] [line]);
    }

    generate Syntax_Null
        '""';

    generate Syntax_Self
        '%self';

    generate Syntax_Indent
    {
        my Unquoted = '    ';
        string Code '"    "';
        defined [my Count]
        {
            Code = (append '$$repeat^galrt("    ",' [my Count Mumps] ')');
            my Unquoted = [null];
        }
        i = Code;
    }

    generate Syntax_Tab
    {
        my Unquoted = [tab];
        string Code '$char(9)';
        defined [my Count]
        {
            Code = (append '$$repeat^galrt($char(9),' [my Count Mumps] ')');
            my Unquoted = [null];
        }
        i = Code;
    }

    generate Syntax_Return
    {
        my Unquoted = (Runtime:Chr 13);
        string Code '$char(13)';
        defined [my Count]
        {
            Code = (append '$$repeat^galrt($char(13),' [my Count Mumps] ')');
            my Unquoted = [null];
        }
        i = Code;
    }

    generate Syntax_Class_Name
    {
        comment `the runtime class is the blessing stored at the handle's
        root`;
        string Target '%self';
        forgive
        {
            string T2 [my Object Mumps];
            if (T2 sgt '')
            {
                Target = T2;
            }
        }
        i = (append '@' Target);
    }

    generate Operation_Is_Null
        '(' Op1 '="")';

    generate Operation_Not_Null
        '(' Op1 "'" '="")';

    generate Operation_Defined
        "(''" '$data(' Variable '))';

    generate Statement_Defined
    {
        comment `python's None is an ABSENT node and '' is a node that
        exists holding nothing - so 'defined' asks $data, not $get. a
        property declared with a value is set by the constructor and is
        defined even when that value is empty; a property declared [null]
        (or with no value at all) is never set, so it is undefined, which
        is exactly what python reports for None. the $etrap absorbs
        errors from the body, the python AttributeError analogue`;
        string H 'L' [my Start_Position] 'H';
        string Guards '';
        object Argument;
        for each Argument in [my Arguments]
        {
            append Guards [indent] "goto:'$data(" [Argument.Mumps] ') ' H [line];
        }
        string Code
            [indent] 'set $etrap="quit:$estack  goto ' H '"' [line]
            Guards
            (i Mumps_Block)
            H ' ; defined' [line]
            [indent] 'set $etrap=' (i Mumps_Trap_Value) ',%ecode=$zstatus,$ecode="",$zstatus=""' [line];
        i = Code;
    }

    generate Statement_Undefined
    {
        string H 'L' [my Start_Position] 'H';
        string Guards '';
        object Argument;
        for each Argument in [my Arguments]
        {
            append Guards [indent] 'goto:$data(' [Argument.Mumps] ') ' H [line];
        }
        string Code
            [indent] 'set $etrap="quit:$estack  goto ' H '"' [line]
            Guards
            (i Mumps_Block)
            H ' ; undefined' [line]
            [indent] 'set $etrap=' (i Mumps_Trap_Value) ',%ecode=$zstatus,$ecode="",$zstatus=""' [line];
        i = Code;
    }

    generate Operation_String_Equal
        '(' Op1 '=' Op2 ')';

    generate Operation_String_Not_Equal
        '(' Op1 "'" '=' Op2 ')';

    generate Operation_String_Greater
        '(' Op1 ']]' Op2 ')';

    generate Operation_String_Less
        '(' Op2 ']]' Op1 ')';

    generate Operation_String_Greater_Equal
        "('(" Op2 ']]' Op1 '))';

    generate Operation_String_Less_Equal
        "('(" Op1 ']]' Op2 '))';

    generate Operation_Contains
        '(' [my String Mumps] '[' [my Search Mumps] ')';

    generate Operation_Append
        '(' (i Mumps_Arguments '_') ')';

    generate Operation_String
        [my Op1 Mumps];

    generate Statement_Break
    {
        i = (append [indent] 'goto ' (i Mumps_Break_Label) [line]);
    }

    generate Statement_Continue
    {
        i = (append [indent] 'goto ' (i Mumps_Continue_Label) [line]);
    }

    generate Statement_Forever
    {
        string Continue 'L' [my Start_Position] 'C';
        string Break 'L' [my Start_Position] 'K';
        string Code

            append Code [indent] 'do lpush^galrt(' Name ',' [Arg.Mumps] ')' [line];
        }
        i = Code;
    }

    generate Statement_Push
    {
        comment `all list mutation goes through the runtime so the count
        node stays truthful and llength is O(1)`;
        string List [my List Mumps];
        string Code '';
        object Arg;
        for each Arg in [my Listargs]
        {
            append Code [indent] 'do lpush^galrt(' List ',' [Arg.Mumps] ')' [line];
        }
        i = Code;
    }

    generate Statement_For_Each
    {
        comment `the gal2 $order walk with position-derived labels, but
        through the runtime cursor pair so one loop shape covers both
        collection kinds: lists yield values in append order, dictionaries
        yield keys in insertion order - the python host contract`;
        string List [my List Mumps];
        string Item [my Item Mumps];
        string Index '%fe' [my Start_Position];
        string Continue 'L' [my Start_Position] 'C';
        string Break 'L' [my Start_Position] 'K';
        string Code [indent] (i Mumps_Declare Index) 'set ' Index '=""' [line]
            Continue ' ; for each ' Item ' continue' [line]
            [indent] 'set ' Index '=$$iternext^galrt(' List ',' Index ')' [line]
            [indent] 'goto:' Index '="" ' Break [line]
            [indent] (i Mumps_Declare Item) 'set ' Item '=$$iterval^galrt(' List ',' Index ')' [line]
            (i Mumps_Block)
            [indent] 'goto ' Continue [line]
            Break ' ; for each ' Item ' break' [line];
        i = Code;
    }

    generate Statement_For_Each_In
    {
        string List [my List Mumps];
        string Item [my Item Mumps];
        string Index '%fe' [my Start_Position];
        string Continue 'L' [my Start_Position] 'C';
        string Break 'L' [my Start_Position] 'K';
        string Code [indent] (i Mumps_Declare Index) 'set ' Index '=""' [line]
            Continue ' ; for each ' Item ' continue' [line]
            [indent] 'set ' Index '=$$iternext^galrt(' List ',' Index ')' [line]
            [indent] 'goto:' Index '="" ' Break [line]
            [indent] (i Mumps_Declare Item) 'set ' Item '=$$iterval^galrt(' List ',' Index ')' [line]
            (i Mumps_Block)
            [indent] 'goto ' Continue [line]
            Break ' ; for each ' Item ' break' [line];
        i = Code;
    }

    generate Statement_Append
    {
        string V [my Variable Mumps];
        string Code [indent] 'set ' V '=' V;
        object Arg;
        for each Arg in [my Listargs]
        {
            append Code '_' [Arg.Mumps];
        }
        append Code [line];
        i = Code;
    }

    generate Statement_Readline
    {
        comment `string literals are prompts, names are input variables,
        mirroring the other backends' readline shape`;
        string Code '';
        string Prompt '';
        string PBetween '';
        objects Arg Argc;
        for each Arg in [my Arguments]
        {
            Argc = [Arg.Classification];
            if (Argc is a Token_Quote)
            {
                append Prompt PBetween [Arg.Mumps];
                PBetween = ',';
            }
            else
            {
                if (Prompt sgt '')
                {
                    append Code [indent] 'use $principal write ' Prompt [line];
                    Prompt = '';
                    PBetween = '';
                }
                append Code [indent] (i Mumps_Declare [Arg.Mumps]) 'read ' [Arg.Mumps] [line];
            }
        }
        i = Code;
    }
    generate Statement_Emit
        [indent] 'use $principal write ' (i Mumps_Arguments ',') [line];
    generate Operation_Ordinal
        '$ascii(' Op1 ')';
    generate Operation_Character
        '$char(' Op1 ')';
    generate Operation_Integer
        '(' Op1 '\1)';
    generate Operation_Power
        '(' Op1 '**' Op2 ')';
}

class Mumps_Compiler is Fallback_Compiler
{
    constructor
    {
        my Generator = (new Mumps_Generator [self]);
    }
    method void Print_Errors
    {
        object E;
        for each E in [my Elements]
        {
            if (not ([E.Error] is null))
            {
                say [line]
                    [class name [E.Classification]]
                    ': '
                    (substring (E.Get_Input) 0 80);
                say 'ERROR: ' [E.Error];
            }
        }
    }
    method number Infer_Code
    {
        verbosity;
        [my Generator Elements] = [my Elements];
        number Status = ([my Generator].Infer_Gal);
        if (not Status)
        {
            i Print_Errors;
            say 'Error in Mumps inference, inferring gal';
            return 0;
        }
        Status = ([my Generator].Infer_Fallback);
        if (not Status)
        {
            i Print_Errors;
            say 'Error in Mumps inference, inferring fallback';
            return 0;
        }
        Status = ([my Generator].Infer_Mumps);
        if (not Status)
        {
            comment `a subset backend fails loudly: a generator error on any
            element must fail the compile, not leak None into the output`;
            i Print_Errors;
            say 'Error in Mumps inference, inferring mumps';
            return 0;
        }
        object Top = (llast [my Elements]);
        try
        {
            if ([Top.Mumps] is null)
            {
                Top.Infer_Mumps;
            }
        }
        catch Error
        {
            i Print_Errors;
            say 'Error in Mumps inference: ' Error;
            return 0;
        }
        my Mumps = [Top.Mumps];
        return 1;
    }
}
