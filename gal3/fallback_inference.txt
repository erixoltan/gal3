class set Spell
{
    method string Fallback_Arguments
    {
        string Code '';
        object Arg;
        for each Arg in [my Arguments]
        {
            append Code ' ' [Arg.Fallback];
        }
        return Code;
    }
    method string Fallback_Args
    {
        string Code '';
        object Arg;
        for each Arg in [my Listargs]
        {
            append Code ' ' [Arg.Fallback];
        }
        return Code;
    }

    method string Append_Fallback_Args
    {
        if ((llength [my Listargs]) > 1)
        {
            return '';
        }
        string Code '';
        object Arg;
        for each Arg in [my Listargs]
        {
            append Code ' ' [Arg.Fallback];
        }
        if ((llength [my Listargs]) > 1)
        {
            return (append ' (append ' Code ')');
        }
        return Code;
    }

    method string Fallback_Memo_Wrapper [string Rule_Name]
    {
        comment `body of the packrat wrapper emitted around every generated
        parse rule. it consults the memo table keyed by token position and
        rule name, replays a cached outcome, or plants an in-progress seed and
        runs the real rule body (renamed with the _Rule suffix) against a
        fresh list. if the body reached its own seed, the rule is left
        recursive: re-run it while the match keeps growing, replaying the
        previous iteration at the recursion point and invalidating dependent
        memo entries each round. replayed elements are the same objects, so
        the element tree is identical to a fresh parse.`;
        string W
            'string Memo_Key (append (string [my Token_Pos]) ":' Rule_Name '");' [line]
            'object Memo_Hit (i Memo_Get Memo_Key);' [line]
            'if (not (Memo_Hit is null))' [line]
            '{' [line]
            [indent] 'return (i Memo_Replay Memo_Hit Parent_Elements);' [line]
            '}' [line]
            'integer Memo_Start_Token [my Token_Pos];' [line]
            'integer Memo_Start_Char [my Char_Pos];' [line]
            'object Memo_Seed (i Memo_Plant Memo_Key);' [line]
            'list Memo_Produced;' [line]
            'flag Memo_Result (i Parse_' Rule_Name '_Rule Memo_Produced);' [line]
            'i Memo_Settle Memo_Seed Memo_Result Memo_Produced;' [line]
            'if ([Memo_Seed.Detected] and Memo_Result)' [line]
            '{' [line]
            [indent] 'forever' [line]
            [indent] '{' [line]
            [indent] [indent] 'i Memo_Regrow_Begin Memo_Seed;' [line]
            [indent] [indent] 'my Token_Pos = Memo_Start_Token;' [line]
            [indent] [indent] 'my Char_Pos = Memo_Start_Char;' [line]
            [indent] [indent] 'list Memo_Grow;' [line]
            [indent] [indent] 'flag Memo_Grow_Result (i Parse_' Rule_Name '_Rule Memo_Grow);' [line]
            [indent] [indent] 'break if (not (i Memo_Grew Memo_Seed Memo_Grow_Result Memo_Grow));' [line]
            [indent] '}' [line]
            '}' [line]
            'i Memo_Finish Memo_Seed;' [line]
            'return (i Memo_Replay Memo_Seed Parent_Elements);' [line];
        return W;
    }
}

generator Fallback Gal
{
    english 'generate the optimal fallback code for each language element in the compiler dialect.';

    generate Element
        [my Gal];

    generate Token_Quote
    {
        string Code (i Enquote [my Unquoted]);
        i = Code;
    }

    generate Statement
    {
        string Code
            [my Verb]
            (i Fallback_Arguments)
            (i Fallback_Block);
        i = Code;
    }
    generate Operation
    {        string Code
            '(' [my Verb] (i Fallback_Arguments) ')';
        i = Code;
    }
    generate Syntax
    {
        string Code [my Verb];
        flag First [true];
        object Arg;
        for each Arg in [my Arguments]
        {
            if (First and (Arg is a Syntax_Tail))
            {
                Code = (i Syntax_Gal Code);
                First = [false];
            }
            append Code ' ' [Arg.Fallback];
        }
        if First
        {
            Code = (i Syntax_Gal Code);
        }
        i = Code;
    }
    generate Syntax_Tail
    {
        join Keys [my Key_Words] ' ';
        string Code Keys (i Fallback_Arguments);
        Code = (i Syntax_Gal Code);
        i = Code;
    }
    generate Block
    {
        string Statements '';
        object Line;
        for each Line in [my Statements]
        {
            append Statements [Line.Fallback] [line];
        }
        string Code;
        if (Statements s= '')
        {
            Code = ';';
        }
        else
        {
            Statements = (i Indent Statements);
            Code = (append [line] '{' [line] Statements '}');
        }
        my Fallback_Statements = Statements;
        i = Code;
    }

    generate Program
    {
        string Code '';
        object Stmt;
        for each Stmt in [my Statements]
        {
            undefined [Stmt.Fallback]
            {
                Stmt.Infer_Fallback;
            }
            Code s+ [Stmt.Fallback] [line];
        }
        i = Code;
    }


    generate Statement_Argument
    {
        string Argument_Name [my Argument_Name Fallback];
        string Code 'property object ' Argument_Name ';';
        string Declaration
            "if ((llength [my Listargs]) = 0)" [line]
            "{" [line] [indent]
                "error 'missing required argument " Argument_Name "';" [line]
            "}" [line]
            'my ' Argument_Name ' = (shift [my Listargs]);' [line];
        append Declaration '[my ' Argument_Name ' Value] = ' [my Data_Type Fallback] ';' [line];
        append Declaration '[my ' Argument_Name ' Usage] = ' [my Usage Fallback] ';' [line];
        my Fallback_Declaration = Declaration;
        i = Code;
    }

    generate Statement_Optional
    {
        string Argument_Name [my Argument_Name Fallback];
        string Code 'property object ' Argument_Name ';';
        string Declaration
            "if ((llength [my Listargs]) > 0)" [line]
            "{" [line] [indent]
                'my ' Argument_Name ' = (shift [my Listargs]);' [line] [indent]
                '[my ' Argument_Name ' Value] = ' [my Data_Type Fallback] ';' [line] [indent]
                '[my ' Argument_Name ' Usage] = ' [my Usage Fallback] ';' [line]
            "}" [line];
        my Fallback_Declaration = Declaration;
        i = Code;
    }

    generate Statement_Flowerbox
    {
        string Comment_Text '';
        object Argument;
        for each Argument in [my Arguments]
        {
            string Arg_Text [Argument.Fallback];
            if (Argument is a Token_Quote)
            {
                Arg_Text = [Argument.Unquoted];
            }
            append Comment_Text Arg_Text;
        }
        list Lines split Comment_Text [line];
        Comment_Text = '';
        strings Line Full;
        integer Width = 0;
        for each Line in Lines
        {
            if ((length Line) > Width)
            {
                Width = (length Line);
            }
        }
        for each Line in Lines
        {
            Full = (append Line '                                                                                                          ');
            Full = (substring Full 0 Width);
            Full = (append '* ' Full ' *');
            append Comment_Text ' ' (i Enquote Full) ' [line]';
        }
        Width += 4;
        string Stars '"' (substring "****************************************************************************************************************" 0 Width) '"';
        Comment_Text = (append Stars ' [line] ' Comment_Text ' ' Stars);
        string Code 'comment ' Comment_Text ';';
        i = Code;
    }
    generate Statement_Method
        'method' (i Fallback_Arguments) (i Fallback_Block);
    generate Statement_Dictionary_Assign
        Dict ' ' Subscript ' d= ' Value ';';

    generate Statement_Generator
    {
        string Language [my Language Fallback];
        string Ancestor 'Generator';
        defined [my Ancestor]
        {
            Ancestor = (append [my Ancestor Fallback] '_Generator');
        }

        string Fallback_Statements '';
        string Fallback_Declaration '';
        object Statement;
        for each Statement in [my Block Statements]
        {
            string Statement_Fb [Statement.Fallback];
            string Statement_Dec [Statement.Fallback_Declaration];
            if (Statement_Dec sgt '')
            {
                append Fallback_Declaration Statement_Dec [line];
            }
            if (Statement_Fb sgt '')
            {
                append Fallback_Statements Statement_Fb [line];
            }
        }
        string Class Language '_Generator';
        string Class_Definition `class ` Class ` is ` Ancestor `
{
    property flag Verbosity [false];
    property string Language_Name ` (i Enquote Language) `;
    property string Property_Name ` (i Enquote Language) `;
    constructor [object Owner [null]]
    {
        my Owner = Owner;
    }
    method flag Infer_` Language `
    {
        comment "this verbosity defined in gal3 fallback_inference.txt for Statement_Generator.";
        verbosity [my Verbosity];
        object Elem;
        object I;
        integer Number = -1;
        variant Error;
        my Status = [true];
        for each Elem in [my Elements]
        {
            try
            {
                Number++;
                if (defined [Elem.Classification])
                {
                    I = [Elem.Classification];
                    I.Infer_` Language `;
                    Elem.` Language ` = [I.` Language `];
                }
                else
                {
                    Elem.Infer_` Language `;
                }
            }
            catch Error
            {
                my Status = [false];
                Elem.Error = Error;
                verbose '!error! ` Language ` element ' Number ': ' [class name I] ', <' Error '>';
            }
        }
        return [false] if (not [my Status]);
        object Top = (llast [my Elements]);
        Top.Infer_` Language `;
        string Code [Top.` Language `];
        my ` Language ` = Code;
        my Code = Code;
        return [true];
    }
}
`;
        string Code
            Fallback_Declaration
            Class_Definition
            Fallback_Statements;
        i = Code;
    }
    generate Statement_Generate
    {
        string Language ([my Parent].Get_Language_Name);
        string Class_Name [my Class_Name Fallback];
        object Argument;
        string Method_Prefix '';
        string Method_Suffix '';
        if ((llength [my Listargs]) > 0)
        {
            append Method_Prefix [indent] 'string Code';
            for each Argument in [my Listargs]
            {
                if ([Argument.Classification] is a Token_Word)
                {
                    append Method_Prefix ' [my ' [Argument.Fallback] ' ' Language ']';
                }
                else
                {
                    append Method_Prefix ' ' [Argument.Fallback];
                }
            }
            append Method_Prefix ';' [line];
            append Method_Suffix [indent] 'my ' Language ' = Code;' [line];
        }
        string Method_Statements
            Method_Prefix
            (i Fallback_Statements)
            Method_Suffix;
        string Method_Code
            'method flag Infer_' Language [line]
            '{' [line]
                Method_Statements
            '}';
        string Code
            'class set ' Class_Name [line]
            '{' [line]
                (i Indent Method_Code)
            [line] '}';
        i = Code;
    }
    generate Statement_Handle
    {
        string Class_Name [my Name Fallback];
        string Inference ([my Parent].Get_Language_Name);
        string Method
            'method flag Infer_' Inference
            (i Fallback_Block);
        string Code
            'class set ' Class_Name [line]
            '{' [line]
                (i Indent Method)
            [line] '}';
        i= Code;
    }

    generate Statement_Parser
    {
        verbosity;
        string Declarations '';
        object Statement;
        forgive
        {
            for each Statement in [my Block Statements]
            {
                defined [Statement.Fallback_Declaration]
                {
                    append Declarations [Statement.Fallback_Declaration];
                }
            }
        }
        string Args (i Fallback_Args);
        string Block (i Fallback_Block);
        string Code
            'class ' [my Name Fallback] '_Parser is Parser'
            Args
            Block
            Declarations;
        my Fallback_Declaration = Declarations;
        i = Code;
    }

    generate Statement_Sequence
    {
        verbosity;
        string Rule_Name [my Rule_Name Fallback];
        string Declarations '';
        flag First [false];
        string Block 'comment ' (i Enquote (i Get_Input)) ';' [line]
            'integer Start_Char = [my Char_Pos];' [line]
            'integer Start_Token = [my Token_Pos];' [line]
            'global list My_Elements;' [line];
        if Verbose
        {
            append Block 'verbosity [true];' [line];
            append Block 'verbose "Parsing sequence ' Rule_Name '.";' [line];
        }
        object Argument;
        flag Excludes [false];
        for each Argument in [my Listargs]
        {
            defined [Argument.Fallback_Declaration]
            {
                append Declarations [Argument.Fallback_Declaration];
            }
            if (Argument is a Token_Quote Syntax_Line Syntax_Tab)
            {
                string Text [Argument.Fallback];
                append Block
                    'if (not (i Parse_Token My_Elements ' Text ')) { i Rollback Start_Char Start_Token (append "' [my Rule_Name Fallback] ' expected "' Text '"."); return [false]; }' [line];
                if Verbose
                {
                    append Block 'verbose [indent] "Sequence ' Rule_Name ' parsed literal: ' Text '";' [line];
                }
            }
            else if (Argument is a Token)
            {
                string Name [Argument.Fallback];
                append Block
                    'if (not (i Parse_' Name ' My_Elements)) { i Rollback Start_Char Start_Token "' [my Rule_Name Fallback] ' expected ' Name '."; return [false]; }' [line];
                if Verbose
                {
                    append Block 'verbose [indent] "Sequence ' Rule_Name ' parsed token ' Name '";' [line];
                }
            }
            else if (Argument is a Syntax_Exclude)
            {
                = Excludes [true];
                append Block [Argument.Fallback] [line];
                if Verbose
                {
                    append Block 'verbose [indent] "Sequence ' Rule_Name ' parsed exclude: ' [Argument.Rule_Name Input] '";' [line];
                }
            }
            else
            {
                append Block [Argument.Fallback] [line];
                if Verbose
                {
                    append Block 'verbose [indent] "Sequence ' Rule_Name ' parsed rule: ' [Argument.Rule_Name Fallback] '";' [line];
                }
            }
        }
        if Verbose
        {
            append Block 'verbose "Sequence ' Rule_Name ' success, start " Start_Char;' [line];
        }
        append Block
            'integer End [my Char_Pos];' [line]
            'object Element (new ' Rule_Name ');' [line]
            'object First_Child (lget [my Tokens] Start_Token);' [line]
            'Start_Char = [First_Child.Start_Position];' [line]
            'Element.Start_Position = Start_Char;' [line]
            'Element.End_Position = End;' [line]
            'Element.Document = [self];' [line]
            'copy [Element.Elements] = My_Elements;' [line]
            'i Add_Element Element;' [line]
            'push Parent_Elements Element;' [line]
            'return [true];' [line];
        if Excludes
        {
            Block = (append 'list Excluded;' [line] Block);
        }

        Block = (i Indent Block);
        string Wrapper (i Indent (i Fallback_Memo_Wrapper Rule_Name));
        string Code 'method flag Parse_' Rule_Name
            ' [list Parent_Elements]' [line]
            '{' [line]
            Wrapper
            '}' [line]
            'method flag Parse_' Rule_Name '_Rule [list Parent_Elements]' [line]
            '{' [line]
            Block
            '}';
        my Fallback_Declarations = Declarations;
        i = Code;
    }

    generate Syntax_Sequence
    {
        debug;
        string Declarations '';
        todo 'the fallback is a reference to the named rule created by this syntax.';
        todo 'the fallback declaration is the sequence function generated by this code.';
        string My_Source (i Get_Input);
        string Rule_Name [my Rule_Name Fallback];
        string Block 'comment ' (i Enquote (i Get_Input)) ';' [line]
            'integer Start_Char [my Char_Pos];' [line]
            'integer Start_Token [my Token_Pos];' [line]
            'global list My_Elements;' [line];
        object Argument;
        flag Excludes [false];
        for each Argument in [my Listargs]
        {
            defined [Argument.Fallback_Declaration]
            {
                append Declarations [Argument.Fallback_Declaration];
            }
            if (Argument is a Token_Quote Syntax_Line)
            {
                string Text [Argument.Fallback];
                append Block
                    'if (not (i Parse_Token My_Elements ' Text ')) { i Rollback Start_Char Start_Token (append "' [my Rule_Name Fallback] ' expected "' Text '"."); return [false]; }' [line];
            }
            else if (Argument is a Token)
            {
                string Name [Argument.Fallback];
                append Block
                    'if (not (i Parse_' Name ' My_Elements)) { i Rollback Start_Char Start_Token "' [my Rule_Name Fallback] ' expected ' Name '."; return [false]; }' [line];
            }
            else if (Argument is a Syntax_Exclude)
            {
                Excludes = [true];
                append Block [Argument.Fallback] [line];
            }
            else
            {
                append Block [Argument.Fallback] [line];
            }
        }
        append Block
            'integer End [my Char_Pos];' [line]
            'object Element = (new ' [my Rule_Name Fallback] ');' [line]
            'object First_Child = (lget [my Tokens] Start_Token);' [line]
            'Start_Char = [First_Child.Start_Position];' [line]
            'Element.Start_Position = Start_Char;' [line]
            'Element.End_Position = End;' [line]
            'Element.Document = [self];' [line]
            'copy [Element.Elements] = My_Elements;' [line]
            'i Add_Element Element;' [line]
            'push Parent_Elements Element;' [line]
            'return [true];' [line];
        if Excludes
        {
            Block = (append 'list Excluded;' [line] Block);
        }

        = Block (i Indent Block);
        string Code
            Declarations
            'method flag Parse_' Rule_Name
            ' [list Parent_Elements]' [line]
            '{' [line]
            Block
            '}';
        my Fallback_Declaration = Code;
        i = Rule_Name;
    }

class Fallback_Compiler is Gal_Compiler
{
    constructor
    {
        my Generator = (new Fallback_Generator [self]);
    }
    method number Infer_Code
    {
        verbosity;
        verbose "Fallback_Compiler.Infer_Code: starting code generation";
        [my Generator Elements] = [my Elements];
        [my Generator].Infer_Gal;
        [my Generator].Verbosity = [true];
        flag Status = ([my Generator].Infer_Fallback);
        return Status if (Status = 0);
        verbose "Fallback_Compiler.Infer_Code: calling Generator.Infer_Fallback";

        # 'Call Infer_Fallback to generate the code';
        [my Generator].Infer_Fallback;

        # 'Get the top-level program element';
        object Top = (llast [my Elements]);

        verbose "Fallback_Compiler.Infer_Code: ensuring Top element has fallback code";

        undefined [Top.Fallback]
        {
            say (i Error_Report);
            return 0;
        }

        string Code = [Top.Fallback];
        my Fallback = Code;

        verbose "Fallback_Compiler.Infer_Code: successfully generated fallback code";

        return 1;
    }
}
