flowerbox 'python to gal translator: classify python elements and emit gal3 source';

comment `
The classification step of the design (design doc section 10). The parse
tree from the generated python parser is grammar-shaped; this file maps it
onto a small semantic tree of Pyc classification nodes, mirroring the gal
compiler's own parser-classification-generator pipeline, and then emits a
gal3 program from that tree. The emitted program uses the python-specific
dialect elements (python print, python while, python for range, and the
py operations) so that every python-semantic decision lives in the gal3
compiler's fallback generators, where a future target language can
override it.

Classification is a single bottom-up pass over the derived element list:
children precede parents, so every element can consult its children's
finished classifications. The default is transparent adoption - an element
with exactly one classified child takes that child's classification, which
collapses grammar plumbing for free. Elements with several classified
children and no specific handler become a Pyc_Group, which either a
handler flattens on purpose or emission rejects with the element's class
name, so unsupported python constructs fail loudly at their own location.

Milestone restrictions (documented in the design doc): module-level
functions only, using parameters and locals; conditions are real
comparisons; string literals are escape-free; assignment targets are plain
names; calls target module functions or the built-ins print and str;
for loops iterate over range with a single argument.
`;

comment 'the classification node classes: a small semantic tree for python';

class Pyc is Element
{
    method string Gal3_Name [string N]
    {
        comment `python names are lowercase but gal names must begin with an
        uppercase letter, an underscore or a dollar sign: names that are
        already valid gal names pass through so machine-generated python
        (whose names came from gal) round-trips readably, the rest get a
        prefix. names the emitted program uses for its own scaffolding are
        always prefixed. method and attribute names are never mapped - they
        live in the attribute namespace, where builtin names such as append
        must survive verbatim.`;
        if ((N s= 'Base') or (N s= 'Py_Program') or (N s= 'Py_Program_Object'))
        {
            return (append 'Py_' N);
        }
        string C (substring N 0 1);
        return N if (contains 'ABCDEFGHIJKLMNOPQRSTUVWXYZ_$' C);
        return (append 'Py_' N);
    }
    method string Gal3_Py_Unescape [string T]
    {
        comment `decode the escape sequences the python literal carries into
        real characters; gal literals do not process escapes, so the real
        characters are re-encoded by Gal3_String_Text`;
        string R '';
        integer P 0;
        integer L (length T);
        forever
        {
            break if (P >= L);
            string C (substring T P 1);
            if (C sne '\')
            {
                append R C;
                increment P;
                continue;
            }
            string E (substring T (P + 1) 1);
            if (E s= 'n')
            {
                append R [line];
            }
            else if (E s= 't')
            {
                append R [tab];
            }
            else if (E s= 'r')
            {
                append R [return];
            }
            else if (E s= '\')
            {
                append R '\';
            }
            else if (E s= "'")
            {
                append R "'";
            }
            else if (E s= '"')
            {
                append R '"';
            }
            else
            {
                error 'pygal: unsupported string escape \' E;
            }
            P = (P + 2);
        }
        return R;
    }
    method string Gal3_Quote_Run [string R]
    {
        if (not (R contains "'"))
        {
            return (append "'" R "'");
        }
        if (not (R contains '"'))
        {
            return (append '"' R '"');
        }
        error 'pygal: string mixes both quote characters';
    }
    method string Gal3_String_Text [string T]
    {
        comment `encode decoded text as a gal literal expression: a plain
        quote when possible, otherwise an append of quoted runs and the
        line and tab literal syntaxes`;
        list Parts;
        string Run '';
        integer P 0;
        integer L (length T);
        forever
        {
            break if (P >= L);
            string C (substring T P 1);
            increment P;
            if (C = [line])
            {
                if (Run sgt '')
                {
                    push Parts (i Gal3_Quote_Run Run);
                    Run = '';
                }
                push Parts '[line]';
                continue;
            }
            if (C = [tab])
            {
                if (Run sgt '')
                {
                    push Parts (i Gal3_Quote_Run Run);
                    Run = '';
                }
                push Parts '[tab]';
                continue;
            }
            if (C = [return])
            {
                if (Run sgt '')
                {
                    push Parts (i Gal3_Quote_Run Run);
                    Run = '';
                }
                push Parts '[return]';
                continue;
            }
            append Run C;
        }
        if (Run sgt '')
        {
            push Parts (i Gal3_Quote_Run Run);
        }
        return "''" if ((llength Parts) = 0);
        return (lget Parts 0) if ((llength Parts) = 1);
        string Code '(append';
        string Q;
        for each Q in Parts
        {
            append Code ' ' Q;
        }
        append Code ')';
        return Code;
    }
    method string Gal3_Block_Of [list Stmts, object Mod, dictionary Scope]
    {
        string B '';
        object S;
        for each S in Stmts
        {
            append B (S.Gal3_Stmt Mod Scope);
        }
        return (append [line] '{' [line] (i Indent B) '}' [line]);
    }
    method string Gal3_Store_Target [object T]
    {
        comment `an attribute chain as an assignment destination: the same
        bracket chain the load form uses`;
        return (T.Gal3_Expr [null]);
    }
    method string Gal3_Stmt [object Mod, dictionary Scope]
    {
        error 'pygal: ' [class name] ' cannot appear as a statement';
    }
    method string Gal3_Expr [object Mod]
    {
        error 'pygal: ' [class name] ' cannot appear as an expression';
    }
}

class Pyc_Attr is Pyc
{
    property object Target;
    property string Attr_Name;
    method flag Is_Self_Attr
    {
        object T [my Target];
        return [false] if (not (T is a Pyc_Name));
        return ([T.Py_Text] s= 'self');
    }
    method string Gal3_Expr [object Mod]
    {
        comment `attribute chains collapse into one bracket expression:
        self.A.B becomes [my A B], X.A.B becomes [X.A B]`;
        string Names [my Attr_Name];
        object T [my Target];
        forever
        {
            break if (not (T is a Pyc_Attr));
            Names = (append [T.Attr_Name] ' ' Names);
            T = [T.Target];
        }
        if (not (T is a Pyc_Name))
        {
            error 'pygal: attribute access on a computed object is not supported';
        }
        if ([T.Py_Text] s= 'self')
        {
            return (append '[my ' Names ']');
        }
        string First '';
        string Rest '';
        integer P 0;
        integer L (length Names);
        forever
        {
            break if (P >= L);
            string C (substring Names P 1);
            increment P;
            if (C s= ' ')
            {
                Rest = (substring Names P (L - P));
                break;
            }
            append First C;
        }
        string Code '[' (i Gal3_Name [T.Py_Text]) '.' First;
        if (Rest sgt '')
        {
            append Code ' ' Rest;
        }
        append Code ']';
        return Code;
    }
}

class Pyc_Method_Call is Pyc
{
    property object Target;
    property string Meth_Name;
    property list Call_Args;
    method string Gal3_Call [object Mod]
    {
        string Args '';
        object A;
        for each A in [my Call_Args]
        {
            append Args ' ' (A.Gal3_Expr Mod);
        }
        object T [my Target];
        if (T is a Pyc_Name)
        {
            if ([T.Py_Text] s= 'self')
            {
                return (append 'i ' [my Meth_Name] Args);
            }
        }
        comment `any expression target works: a name, an attribute chain,
        a string literal for join, a subscript`;
        return (append (T.Gal3_Expr Mod) '.' [my Meth_Name] Args);
    }
    method string Gal3_Stmt [object Mod, dictionary Scope]
    {
        object T [my Target];
        if (T is a Pyc_Name)
        {
            if ([T.Py_Text] s= 'pdebug')
            {
                if ([my Meth_Name] s= 'zbreak')
                {
                    comment 'the compiled form of the debug statement';
                    return (append 'debug;' [line]);
                }
            }
        }
        return (append (i Gal3_Call Mod) ';' [line]);
    }
    method string Gal3_Expr [object Mod]
    {
        return (append '(' (i Gal3_Call Mod) ')');
    }
}

class Pyc_Classdef is Pyc
{
    property string Py_Cname;
    property string Base_Name '';
    property list Methods;
    property list Prop_Lines;
    method string Gal3_Class [object Mod]
    {
        string CN (i Gal3_Name [my Py_Cname]);
        string Code 'class ' CN;
        if ([my Base_Name] sgt '')
        {
            append Code ' is ' (i Gal3_Name [my Base_Name]);
        }
        if (((llength [my Methods]) = 0) and ((llength [my Prop_Lines]) = 0))
        {
            append Code ';' [line];
            return Code;
        }
        append Code [line] '{' [line];
        string Ms '';
        string PL;
        for each PL in [my Prop_Lines]
        {
            append Ms PL [line];
        }
        Mod.In_Method = [true];
        object M;
        for each M in [my Methods]
        {
            append Ms (M.Gal3_Method Mod);
        }
        Mod.In_Method = [false];
        append Code (i Indent Ms) '}' [line];
        return Code;
    }
}

    {

        string Code '(py fstr';

        append Code ')';
        return Code;
    }

class Pyc_Module is Pyc
{
    property list Stmts;
    property dictionary Funcs;
    property dictionary Classes;
    property object Compiler;
    property flag In_Method [false];
    property list Main_Params;

    method flag Pyg_Take_Main_Params [list Tops]
    {
        comment `the compiled main-arguments idiom: a leading try whose body
        assigns names from sys.argv and whose handler prints usage and calls
        sys.exit. the names become main's parameters.`;
        return [false] if ((llength Tops) = 0);
        comment `elided imports precede the idiom as pass markers`;
        integer TI 0;
        integer TN (llength Tops);
        forever
        {
            return [false] if (TI >= TN);
            object TS (lget Tops TI);
            break if (not (TS is a Pyc_Pass));
            increment TI;
        }
        object T0 (lget Tops TI);
        return [false] if (not (T0 is a Pyc_Try));
        list Names;
        object S;
        for each S in [T0.Body]
        {
            return [false] if (not (S is a Pyc_Assign));
            object TN [S.Target];
            return [false] if (not (TN is a Pyc_Name));
            object V [S.Value];
            return [false] if (not (V is a Pyc_Binop));
            return [false] if (not ([V.Verb] s= 'py sub'));
            object L [V.Left];
            return [false] if (not (L is a Pyc_Attr));
            object LT [L.Target];
            return [false] if (not (LT is a Pyc_Name));
            return [false] if (not ([LT.Py_Text] s= 'sys'));
            return [false] if (not ([L.Attr_Name] s= 'argv'));
            push Names [TN.Py_Text];
        }
        return [false] if ((llength Names) = 0);
        flag Exits [false];
        for each S in [T0.Catch_Body]
        {
            forgive
            {
                if (S is a Pyc_Method_Call)
                {
                    object MT [S.Target];
                    if (MT is a Pyc_Name)
                    {
                        if ([MT.Py_Text] s= 'sys')
                        {
                            Exits = [true];
                        }
                    }
                }
            }
        }
        return [false] if (not Exits);
        my Main_Params = Names;
        list Keep;
        integer KI 0;
        object KS;
        for each KS in Tops
        {
            if (KI != TI)
            {
                push Keep KS;
            }
            increment KI;
        }
        lclear Tops;
        for each KS in Keep
        {
            push Tops KS;
        }
        return [true];
    }

    method flag Is_Main_Guard [object S]
    {
        comment `recognize if __name__ == '__main__': its suite is the
        program body`;
        return [false] if (not (S is a Pyc_If));
        return [false] if (not ([S.Else_Part] is null));
        object C [S.Cond];
        return [false] if (not (C is a Pyc_Binop));
        return [false] if (not ([C.Verb] s= 'py ='));
        object L [C.Left];
        return [false] if (not (L is a Pyc_Name));
        return ([L.Py_Text] s= '__name__');
    }

    method string Gal3_Program
    {
        list Defs;
        list Classes_L;
        list Tops;
        object S;
        for each S in [my Stmts]
        {
            if (S is a Pyc_Funcdef)
            {
                push Defs S;
            }
            else if (S is a Pyc_Classdef)
            {
                push Classes_L S;
            }
            else if (i Is_Main_Guard S)
            {
                object T2 [S.Then_Suite];
                object M2;
                for each M2 in [T2.Stmts]
                {
                    push Tops M2;
                }
            }
            else
            {
                push Tops S;
            }
        }
        comment `register every module function and class before emitting
        any body, so forward references resolve`;
        for each S in Defs
        {
            string K [S.Py_Fname];
            [my Funcs] K d= (i Gal3_Name K);
        }
        for each S in Classes_L
        {
            string K2 [S.Py_Cname];
            [my Classes] K2 d= (i Gal3_Name K2);
        }
        comment `the same argument-sink Base the compiler programs get
        from compiler_base: constructors chain to it with their own
        arguments, so it must absorb them`;
        comment `the whole program lives inside a module: main arguments
        and dynamic includes both need the module context`;
        string Code "flowerbox 'translated from python by pygal';" [line]
            'module Compiler' [line]
            '{' [line];
        string Inner 'class Base' [line]
            '{' [line]
            '    constructor [variant A [null], variant B [null], variant C [null]];' [line]
            '}' [line];
        for each S in Classes_L
        {
            append Inner (S.Gal3_Class [self]);
        }
        if ((llength Defs) > 0)
        {
            append Inner 'class Py_Program' [line] '{' [line];
            my In_Method = [true];
            string Methods '';
            for each S in Defs
            {
                append Methods (S.Gal3_Method [self]);
            }
            my In_Method = [false];
            append Inner (i Indent Methods) '}' [line];
        }
        flag Has_Args (i Pyg_Take_Main_Params Tops);
        append Inner 'main';
        dictionary Scope;
        if Has_Args
        {
            string MP;
            flag MFirst [true];
            for each MP in [my Main_Params]
            {
                Scope MP d= [true];
                if MFirst
                {
                    append Inner ' [string ' (i Gal3_Name MP);
                    MFirst = [false];
                }
                else
                {
                    append Inner ', string ' (i Gal3_Name MP);
                }
            }
            append Inner ']';
        }
        append Inner [line] '{' [line];
        string Body '';
        if ((llength Defs) > 0)
        {
            append Body 'object Py_Program_Object = (new Py_Program);' [line];
        }
        for each S in Tops
        {
            append Body (S.Gal3_Stmt [self] Scope);
        }
        append Inner (i Indent Body) '}' [line];
        append Code (i Indent Inner) '}' [line];
        return Code;
    }
}

class set Element
{
    property object Pyc [null];

    method flag Pyg_Is_Py_Keyword [string T]
    {
        string K ' and as assert async await break class continue def del elif else except finally for from global if import in is lambda nonlocal not or pass raise return try while with yield ';
        return (contains K (append ' ' T ' '));
    }

    method void Pyg_Flatten [object P, list Into]
    {
        return if (P is null);
        if (P is a Pyc_Group)
        {
            object M;
            for each M in [P.Items]
            {
                i Pyg_Flatten M Into;
            }
            return;
        }
        push Into P;
    }

    method void Pyg_Kids [list Into]
    {
        comment `the classified children in order, with grammar-plumbing
        groups flattened away`;
        object Child;
        for each Child in [my Elements]
        {
            i Pyg_Flatten [Child.Pyc] Into;
        }
    }

    method object Pyc_Build
    {
        return (i Pyc_Build_Default);
    }

    method object Pyc_Build_Default
    {
        comment `the default: tokens classify to nothing, one classified
        child is adopted transparently, several become a group that a
        specific handler must consume. interior synthetics inherit their
        rule's class, so rule handlers guard on the exact class name and
        fall back here for their interiors.`;
        return [null] if ([self] is a Token);
        list Kids;
        object Child;
        for each Child in [my Elements]
        {
            continue if ([Child.Pyc] is null);
            push Kids [Child.Pyc];
        }
        integer N (llength Kids);
        return [null] if (N = 0);
        return (lget Kids 0) if (N = 1);
        object G (new Pyc_Group);
        G.Items = Kids;
        G.From_Class = [class name];
        return G;
    }

    method object Pyg_Build_Binop [string Verb]
    {
        list Kids;
        i Pyg_Kids Kids;
        if ((llength Kids) != 2)
        {
            error 'pygal: ' [class name] ' expected two operands';
        }
        object B (new Pyc_Binop);
        B.Verb = Verb;
        B.Left = (lget Kids 0);
        B.Right = (lget Kids 1);
        return B;
    }

    method object Pyg_Build_Unop [string Verb]
    {
        list Kids;
        i Pyg_Kids Kids;
        if ((llength Kids) != 1)
        {
            error 'pygal: ' [class name] ' expected one operand';
        }
        object U (new Pyc_Unop);
        U.Verb = Verb;
        U.Operand = (lget Kids 0);
        return U;
    }

    method object Pyg_Build_Tail [string Verb]
    {
        list Kids;
        i Pyg_Kids Kids;
        if ((llength Kids) != 1)
        {
            error 'pygal: ' [class name] ' expected one operand';
        }
        object T (new Pyc_Op_Tail);
        T.Verb = Verb;
        T.Operand = (lget Kids 0);
        return T;
    }

    method object Pyg_Build_Fold
    {
        comment `left fold for chains of one associative operator:
        a or b or c nests as ((a or b) or c)`;
        list Kids;
        i Pyg_Kids Kids;
        object Acc (shift Kids);
        object T;
        for each T in Kids
        {
            if (not (T is a Pyc_Op_Tail))
            {
                error 'pygal: ' [class name] ' malformed operator chain';
            }
            object B (new Pyc_Binop);
            B.Verb = [T.Verb];
            B.Left = Acc;
            B.Right = [T.Operand];
            Acc = B;
        }
        return Acc;
    }

    method flag Pyg_Is_Super_Init [object S]
    {
        return [false] if (not (S is a Pyc_Method_Call));
        return [false] if (not ([S.Meth_Name] s= '__init__'));
        object T [S.Target];
        return [false] if (not (T is a Pyc_Call));
        return ([T.Callee_Name] s= 'super');
    }

    method object Pyg_Build_Attr
    {
        comment `an attribute access: the target and the attribute name.
        the name stays verbatim - it lives in the attribute namespace.`;
        list Kids;
        i Pyg_Kids Kids;
        if ((llength Kids) != 2)
        {
            error 'pygal: ' [class name] ' unexpected attribute shape';
        }
        object Nm (lget Kids 1);
        if (not (Nm is a Pyc_Name))
        {
            error 'pygal: attribute name did not classify to a name';
        }
        object A (new Pyc_Attr);
        A.Target = (lget Kids 0);
        A.Attr_Name = [Nm.Py_Text];
        return A;
    }

    method object Pyg_Build_Subscript
    {
        list Kids;
        i Pyg_Kids Kids;
        if ((llength Kids) != 2)
        {
            error 'pygal: ' [class name] ' unexpected subscript shape';
        }
        object Idx (lget Kids 1);
        if (Idx is a Pyc_SliceLit)
        {
            object S3 (new Pyc_Slice3);
            S3.Target = (lget Kids 0);
            S3.Lower = [Idx.Lower];
            S3.Upper = [Idx.Upper];
            return S3;
        }
        object B (new Pyc_Binop);
        B.Verb = 'py sub';
        B.Left = (lget Kids 0);
        B.Right = Idx;
        return B;
    }

    method object Pyg_Guard_Arg [object S]
    {
        comment `recognize one guard line of the defined idiom:
        if X is None: raise ValueError -- returns X or null`;
        return [null] if (not (S is a Pyc_If));
        return [null] if (not ([S.Else_Part] is null));
        object C [S.Cond];
        return [null] if (not (C is a Pyc_Binop));
        return [null] if (not ([C.Verb] s= 'py is'));
        object R [C.Right];
        return [null] if (not (R is a Pyc_Name));
        return [null] if (not ([R.Py_Text] s= 'None'));
        object TS [S.Then_Suite];
        return [null] if ((llength [TS.Stmts]) != 1);
        object RS (lget [TS.Stmts] 0);
        return [null] if (not (RS is a Pyc_Raise));
        object EX [RS.Exc];
        return [null] if (not (EX is a Pyc_Name));
        return [null] if (not ([EX.Py_Text] s= 'ValueError'));
        return [C.Left];
    }

    method object Pyg_As_Suite [object P]
    {
        comment `a same-line suite such as except Exception: pass has no
        block element, so the single statement arrives bare`;
        return P if (P is a Pyc_Suite);
        object S (new Pyc_Suite);
        push [S.Stmts] P;
        return S;
    }

    method flag Pyg_Is_Veae [object T]
    {
        comment 'the (ValueError, AttributeError) tuple of the defined idiom';
        return [false] if (not (T is a Pyc_TupleLit));
        return [false] if ((llength [T.Items]) != 2);
        flag Ve [false];
        flag Ae [false];
        object N;
        for each N in [T.Items]
        {
            continue if (not (N is a Pyc_Name));
            if ([N.Py_Text] s= 'ValueError')
            {
                Ve = [true];
            }
            if ([N.Py_Text] s= 'AttributeError')
            {
                Ae = [true];
            }
        }
        return (Ve and Ae);
    }

    method void Pyg_Rewrite_Stmts [list Stmts]
    {
        comment `recover multi-statement compiled idioms. the file-read
        triple: handle = open(F, ...), var = handle.read(), handle.close()
        becomes one fread statement.`;
        integer N (llength Stmts);
        integer P 0;
        list Out;
        forever
        {
            break if (P >= N);
            flag Matched [false];
            if ((P + 2) < N)
            {
                object S0 (lget Stmts P);
                object S1 (lget Stmts (P + 1));
                object S2 (lget Stmts (P + 2));
                forgive
                {
                    if ((S0 is a Pyc_Assign) and (S1 is a Pyc_Assign) and (S2 is a Pyc_Method_Call))
                    {
                        object H0 [S0.Target];
                        object V0 [S0.Value];
                        object T1 [S1.Target];
                        object V1 [S1.Value];
                        if ((H0 is a Pyc_Name) and (V0 is a Pyc_Call) and (T1 is a Pyc_Name) and (V1 is a Pyc_Method_Call))
                        {
                            if (([V0.Callee_Name] s= 'open') and ((llength [V0.Call_Args]) > 0))
                            {
                                object RT [V1.Target];
                                object CT [S2.Target];
                                if ((RT is a Pyc_Name) and (CT is a Pyc_Name))
                                {
                                    if (([RT.Py_Text] s= [H0.Py_Text]) and ([CT.Py_Text] s= [H0.Py_Text])
                                        and ([V1.Meth_Name] s= 'read') and ([S2.Meth_Name] s= 'close'))
                                    {
                                        object FR (new Pyc_Fread);
                                        FR.Var_Name = [T1.Py_Text];
                                        FR.File_Expr = (lget [V0.Call_Args] 0);
                                        push Out FR;
                                        P = (P + 3);
                                        Matched = [true];
                                    }
                                }
                            }
                        }
                    }
                }
            }
            continue if Matched;
            push Out (lget Stmts P);
            increment P;
        }
        lclear Stmts;
        object O;
        for each O in Out
        {
            push Stmts O;
        }
    }

    method object Pyg_Build_Suite
    {
        object S (new Pyc_Suite);
        i Pyg_Kids [S.Stmts];
        i Pyg_Rewrite_Stmts [S.Stmts];
        return S;
    }

    method object Pyg_Build_If
    {
        list Kids;
        i Pyg_Kids Kids;
        integer N (llength Kids);
        if ((N < 2) or (N > 3))
        {
            error 'pygal: ' [class name] ' unexpected if statement shape';
        }
        object F (new Pyc_If);
        F.Cond = (lget Kids 0);
        F.Then_Suite = (lget Kids 1);
        if (N = 3)
        {
            F.Else_Part = (lget Kids 2);
        }
        return F;
    }

    method object Pyg_Build_Def
    {
        comment `gathered children of a function definition: the name, then
        one name per parameter, then the body suite`;
        list Kids;
        i Pyg_Kids Kids;
        integer N (llength Kids);
        if (N < 2)
        {
            error 'pygal: ' [class name] ' unexpected function definition shape';
        }
        object Name (lget Kids 0);
        object Body (llast Kids);
        if (not (Name is a Pyc_Name))
        {
            error 'pygal: function name did not classify to a name';
        }
        if (not (Body is a Pyc_Suite))
        {
            error 'pygal: function body did not classify to a suite; annotations and defaults are not supported';
        }
        object F (new Pyc_Funcdef);
        F.Py_Fname = [Name.Py_Text];
        F.Body = Body;
        integer End (N - 2);
        integer Sub;
        for Sub range 1 End
        {
            object P (lget Kids Sub);
            if (P is a Pyc_Default)
            {
                comment `a default rides with the parameter before it; only
                the None default has a gal spelling so far`;
                if ((llength [F.Param_Names]) = 0)
                {
                    error 'pygal: parameter default without a parameter';
                }
                object DV [P.Value];
                if (not (DV is a Pyc_Name))
                {
                    error 'pygal: only None parameter defaults are supported';
                }
                if (not ([DV.Py_Text] s= 'None'))
                {
                    error 'pygal: only None parameter defaults are supported';
                }
                object Drop (pop [F.Param_Defaults]);
                push [F.Param_Defaults] ' [null]';
                continue;
            }
            if (not (P is a Pyc_Name))
            {
                error 'pygal: parameter did not classify to a plain name; annotations are not supported';
            }
            push [F.Param_Names] [P.Py_Text];
            push [F.Param_Defaults] '';
        }
        return F;
    }
}

class set Py_String
{
    method object Pyc_Build
    {
        string T [my Text];
        integer L (length T);
        integer Q 0;
        string C '';
        forever
        {
            break if (Q >= L);
            C = (substring T Q 1);
            break if ((C s= "'") or (C s= '"'));
            increment Q;
        }

        flag Is_F [false];
        flag Is_Raw [false];

            if ((PC s= 'f') or (PC s= 'F'))
            {
                Is_F = [true];
            }
            else if ((PC s= 'r') or (PC s= 'R'))
            {
                Is_Raw = [true];
            }
            else
            {
                error 'pygal: unsupported string prefix ' Prefix;
            }
        }
        integer Qlen 1;
        string Q3 (substring T Q 3);
        if ((Q3 s= "'''") or (Q3 s= '"""'))
        {
            Qlen = 3;
        }
        integer IStart (Q + Qlen);
        integer ILen ((L - IStart) - Qlen);
        string Interior (substring T IStart ILen);
        if (not Is_F)
        {
            object P (new Pyc_String);
            P.Interior = Interior;
            P.No_Unescape = Is_Raw;
            return P;
        }
        if Is_Raw
        {
            error 'pygal: raw f-strings are not supported';
        }
        comment `split the f-string into literal runs and expression
        segments; expressions are parsed at emission time`;
        object F (new Pyc_Fstring);
        string Run '';
        integer P2 0;
        integer IL (length Interior);
        forever
        {
            break if (P2 >= IL);
            string C2 (substring Interior P2 1);
            string C22 (substring Interior P2 2);
            if (C22 s= '{{')
            {
                append Run '{';
                P2 = (P2 + 2);
                continue;
            }
            if (C22 s= '}}')
            {
                append Run '}';
                P2 = (P2 + 2);
                continue;
            }
            if (C2 s= '{')
            {
                if (Run sgt '')
                {
                    object Sg (new Pyc_Fseg);
                    Sg.Seg_Text = Run;
                    push [F.Parts] Sg;
                    Run = '';
                }
                integer E2 (P2 + 1);
                string EC '';
                string Ex '';
                integer Depth 0;
                flag In_Q [false];
                string QC '';
                forever
                {
                    if (E2 >= IL)
                    {
                        error 'pygal: unterminated f-string expression';
                    }
                    EC = (substring Interior E2 1);
                    if In_Q
                    {
                        if (EC s= QC)
                        {
                            In_Q = [false];
                        }
                        append Ex EC;
                        increment E2;
                        continue;
                    }
                    if ((EC s= "'") or (EC s= '"'))
                    {
                        In_Q = [true];
                        QC = EC;
                        append Ex EC;
                        increment E2;
                        continue;
                    }
                    if ((EC s= '(') or (EC s= '['))
                    {
                        increment Depth;
                    }
                    else if ((EC s= ')') or (EC s= ']'))
                    {
                        Depth = (Depth - 1);
                    }
                    else if (Depth = 0)
                    {
                        break if (EC s= '}');
                        comment `a colon at bracket depth zero would start a
                        format specification, which nothing generates`;
                        if ((EC s= '{') or (EC s= ':'))
                        {
                            error 'pygal: unsupported f-string expression form';
                        }
                    }
                    append Ex EC;
                    increment E2;
                }
                object Fx (new Pyc_Fexpr);
                Fx.Expr_Text = Ex;
                push [F.Parts] Fx;
                P2 = (E2 + 1);
                continue;
            }
            if (C2 s= '}')
            {
                error 'pygal: stray closing brace in f-string';
            }
            append Run C2;
            increment P2;
        }
        if (Run sgt '')
        {
            object Sg2 (new Pyc_Fseg);
            Sg2.Seg_Text = Run;
            push [F.Parts] Sg2;
        }
        return F;
    }

class set Python_try_stmt_A151
{
    method object Pyc_Build
    {
        comment `one try, exactly one handler, no else or finally - the only
        shape gal generates. three gal statements come home from here: the
        defined and undefined idioms (guarded ValueError raises with an
        except (ValueError, AttributeError) handler) and forgive (a pass
        handler); everything else is a plain try/catch.`;
        list Kids;
        i Pyg_Kids Kids;
        if ((llength Kids) != 2)
        {
            string Diag '';
            object DK;
            for each DK in Kids
            {
                append Diag ' ' [class DK name];
            }
            error 'pygal: only try with exactly one except block is supported, got' Diag;
        }
        object TB (lget Kids 0);
        object H (lget Kids 1);
        if (not (TB is a Pyc_Suite))
        {
            error 'pygal: try body did not classify to a suite';
        }
        if (not (H is a Pyc_Except))
        {
            error 'pygal: try handler did not classify to an except block';
        }
        object HB [H.Body];
        flag HB_Pass [false];
        if ((llength [HB.Stmts]) = 1)
        {
            object H0 (lget [HB.Stmts] 0);
            if (H0 is a Pyc_Pass)
            {
                HB_Pass = [true];
            }
        }
        object HT [H.Exc_Type];
        if (i Pyg_Is_Veae HT)
        {
            if ((llength [TB.Stmts]) < 1)
            {
                error 'pygal: unrecognized guarded try idiom';
            }
            object G0 (lget [TB.Stmts] 0);
            object Arg (i Pyg_Guard_Arg G0);
            if (Arg is null)
            {
                error 'pygal: unrecognized guarded try idiom';
            }
            list Rest;
            integer Sub;
            integer End ((llength [TB.Stmts]) - 1);
            for Sub range 1 End
            {
                object RG (i Pyg_Guard_Arg (lget [TB.Stmts] Sub));
                if (not (RG is null))
                {
                    error 'pygal: multiple defined guards are not supported yet';
                }
                push Rest (lget [TB.Stmts] Sub);
            }
            if HB_Pass
            {
                object DF (new Pyc_Defined);
                DF.Arg = Arg;
                DF.Body = Rest;
                return DF;
            }
            if ((llength Rest) = 0)
            {
                object UD (new Pyc_Undefined);
                UD.Arg = Arg;
                UD.Body = [HB.Stmts];
                return UD;
            }
            error 'pygal: unrecognized guarded try idiom';
        }
        flag Catchall [false];
        if (HT is null)
        {
            Catchall = [true];
        }
        else if (HT is a Pyc_Name)
        {
            if ([HT.Py_Text] s= 'Exception')
            {
                Catchall = [true];
            }
        }
        if (not Catchall)
        {
            error 'pygal: only except Exception handlers are supported';
        }
        if (HB_Pass and ([H.Err_Name] s= ''))
        {
            object FG (new Pyc_Forgive);
            FG.Body = [TB.Stmts];
            return FG;
        }
        object TR (new Pyc_Try);
        TR.Body = [TB.Stmts];
        TR.Err_Name = [H.Err_Name];
        TR.Catch_Body = [HB.Stmts];
        return TR;
    }
}

class Pygal_Compiler is Py_Parse_Compiler
{
    property string Gal3;

    method object Pyg_Expression_Pyc [string T]
    {
        comment `parse one python expression - the interior of an f-string
        segment - with a child compiler, classify its elements, and return
        the root classification. the padding of the generated { expr } form
        must go, or the sub-lexer reads it as indentation.`;
        forever
        {
            integer L2 (length T);
            break if (L2 = 0);
            string C2 (substring T 0 1);
            break if ((C2 sne ' ') and (C2 sne [tab]));
            T = (substring T 1 (L2 - 1));
        }
        forever
        {
            integer L3 (length T);
            break if (L3 = 0);
            string C3 (substring T (L3 - 1) 1);
            break if ((C3 sne ' ') and (C3 sne [tab]));
            T = (substring T 0 (L3 - 1));
        }
        object C (new Pygal_Compiler);
        C.Text = (append T [line]);
        if ((C.Tokenize) = 0)
        {
            error 'pygal: cannot tokenize f-string expression: ' T;
        }
        object P [C.Parser];
        P.Text = [C.Text];
        P.Tokens = [C.Tokens];
        list Elems;
        number Status (P.Parse_Python_expression Elems);
        if (Status = 0)
        {
            error 'pygal: cannot parse f-string expression: ' T;
        }
        if ((llength Elems) = 0)
        {
            error 'pygal: f-string expression produced no elements: ' T;
        }
        comment `the winning root is what the parse pushed into Elems; the
        parser's own element list also holds orphans from discarded
        left-recursion grow iterations, so the flat list is derived by
        walking the winning tree`;
        object Root (llast Elems);
        list Ordered;
        C.Walk_Element Root Ordered;
        C.Elements = Ordered;
        object Elem;
        for each Elem in [C.Elements]
        {
            Elem.Pyc = (Elem.Pyc_Build);
        }
        object R [Root.Pyc];
        if (R is null)
        {
            error 'pygal: f-string expression classified to nothing: ' T;
        }
        return R;
    }

    method number Translate
    {
        verbosity;
        return 0 if ((i Parse_Python) = 0);
        verbose 'pygal: classifying ' (string (llength [my Elements])) ' elements';
        object Elem;
        for each Elem in [my Elements]
        {
            Elem.Pyc = (Elem.Pyc_Build);
        }
        object Root (llast [my Elements]);
        object M [Root.Pyc];
        if (not (M is a Pyc_Module))
        {
            say 'pygal: the program root did not classify to a module';
            return 0;
        }
        M.Compiler = [self];
        my Gal3 = (M.Gal3_Program);
        return 1;
    }
}
